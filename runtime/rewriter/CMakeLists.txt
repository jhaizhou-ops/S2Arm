cmake_minimum_required(VERSION 3.18)
project(s2a_rewriter C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(S2A_REWRITER_DEPS "" CACHE PATH "prepare_deps.py 生成的依赖目录（离线构建）")
if(NOT EXISTS "${S2A_REWRITER_DEPS}/capstone-5.0.7/src/CMakeLists.txt" OR
   NOT EXISTS "${S2A_REWRITER_DEPS}/lz4-4.4.5/lz4libs/lz4.c")
  message(FATAL_ERROR "先显式运行 prepare_deps.py <依赖目录>；构建和测试不自动联网")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CAPSTONE_BUILD_MACOS_THIN ON CACHE BOOL "" FORCE)
set(CAPSTONE_ARCHITECTURE_DEFAULT OFF CACHE BOOL "" FORCE)
set(CAPSTONE_ARM64_SUPPORT ON CACHE BOOL "" FORCE)
set(CAPSTONE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CAPSTONE_BUILD_CSTOOL OFF CACHE BOOL "" FORCE)
set(CAPSTONE_BUILD_CSTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${S2A_REWRITER_DEPS}/capstone-5.0.7/src" capstone)
add_library(s2a_rewriter STATIC rewriter.c "${S2A_REWRITER_DEPS}/lz4-4.4.5/lz4libs/lz4.c")
set_target_properties(s2a_rewriter PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(s2a_rewriter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(s2a_rewriter PRIVATE "${S2A_REWRITER_DEPS}/lz4-4.4.5/lz4libs")
target_link_libraries(s2a_rewriter PUBLIC capstone)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(s2a_rewriter PRIVATE -Wall -Wextra)
endif()
add_executable(s2a-rewrite main.c)
target_link_libraries(s2a-rewrite PRIVATE s2a_rewriter)
option(S2A_REWRITER_API_TEST "构建宿主 C API 并发测试" OFF)
if(S2A_REWRITER_API_TEST)
  find_package(Threads REQUIRED)
  add_executable(s2a-rewrite-api-test ../../tests/rewriter_api_test.c)
  target_link_libraries(s2a-rewrite-api-test PRIVATE s2a_rewriter Threads::Threads)
endif()
